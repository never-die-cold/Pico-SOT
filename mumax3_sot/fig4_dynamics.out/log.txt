//output directory: fig4_dynamics.out/
SetGridSize(64, 64, 1)
SetCellSize(16e-9, 16e-9, 1e-9)
// WARNING: cell size was set to a high aspect ratio.
//          Calculation of demag kernel may take longer than usual.
//          Note that this can exacerbate discretization anisotropy.
SetPBC(16, 16, 0)
//resizing...
Ms0 := 1.3e6
Msat = Ms0
Aex = 3e-11
alpha = 0.1
Ha0 := 1.0
Ku0 := 0.5*Ms0*Ha0 + 0.5*mu0*Ms0*Ms0
Ku1 = Ku0
AnisU = vector(0, 0, 1)
DisableZhangLiTorque = true
Pol = 0.20
EpsilonPrime = 0.05
Lambda = 1
FreeLayerThickness = 1e-9
FixedLayer = vector(0, 1, 0)
J = vector(0, 0, 0)
Hx := 160e-3
Isign := 1.0
B_ext = vector(Hx, 0, 0)
Jpk := 4.0e12
tp := 3.7e-12
t0 := 5e-12
dTpk := 40.0
tauC := 60e-12
Troom := 300.0
Tc := 800.0
echo := 0.0
ted := 24e-12
ln2x4 := 2.772588722239781
SetSolver(4)
FixDt = 5e-14
Tcur := Troom
Jcur := 0.0
Ms_cur := Ms0
TableAddVar(Tcur, "T", "K")
TableAddVar(Jcur, "J", "A/m^2")
TableAddVar(Ms_cur, "Ms", "A/m")
TableAutoSave(0.25e-12)
OutputFormat = OVF2_BINARY
m = Uniform(0, 0, 1)
relax()
saveas(m, "m_initial")
print("relaxed m = ", m.average(), "  B_demag = ", B_demag.average())
//relaxed m =  [0.1596541553735733 9.579935733583511e-10 0.9871730208396912]   B_demag =  [-1.331971725448966e-05 -7.882919577512082e-14 -1.610617756843567]
tmax := 400e-12
stp := 0.25e-12
for tt := 0.0; tt < tmax; tt += stp {
	gg := exp(-ln2x4*pow((tt-t0)/tp, 2)) + echo*exp(-ln2x4*pow((tt-t0-ted)/tp, 2))
	Jcur = Isign * Jpk * gg
	J = vector(0, 0, Jcur)
	heat := dTpk * gg * gg / (0.75 * tp)
	Tcur = (Tcur + heat*stp + Troom*stp/tauC) / (1 + stp/tauC)
	frac := (Tc - Tcur) / (Tc - Troom)
	frac = 0.5 * (frac + abs(frac))
	Ms_cur = Ms0 * pow(frac, 0.5)
	Msat = Ms_cur
	Ku1 = Ku0 * frac
	run(stp)
}
J = vector(0, 0, 0)
Jcur = 0.0
tableSave()
saveas(m, "m_final")
print("final m = ", m.average())
//final m =  [0.15954984724521637 -3.7826834159204736e-05 0.9871898293495178]
