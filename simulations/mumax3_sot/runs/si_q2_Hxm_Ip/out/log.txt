//output directory: E:\_SOT_MARM\simulations\mumax3_sot\runs\si_q2_Hxm_Ip\out/
Nx := 500
Ny := 400
SetGridSize(Nx, Ny, 1)
SetCellSize(5e-6/Nx, 4e-6/Ny, 1e-9)
// WARNING: cell size was set to a high aspect ratio.
//          Calculation of demag kernel may take longer than usual.
//          Note that this can exacerbate discretization anisotropy.
Ms300 := 1.0e6
BK300 := 0.8
Kz300 := 0.5 * Ms300 * (BK300 + mu0*Ms300)
Troom := 300.0
Tc := 800.0
fr300 := 1.0 - pow(Troom/Tc, 1.7)
Ms0 := Ms300 / fr300
Kz0 := Kz300 / pow(fr300, 3.0)
Msat = Ms300
Aex = 3e-11
alpha = 0.23
Ku1 = Kz300
AnisU = vector(0, 0, 1)
DisableZhangLiTorque = true
Pol = 0.20
EpsilonPrime = 0.05
Lambda = 1
FreeLayerThickness = 1e-9
FixedLayer = vector(0, -1, 0)
J = vector(0, 0, 0)
Hx := -0.160
Isign := 1
B_ext = vector(Hx, 0, 0)
Jpk := 1.2e13
tp := 6.0e-12
t0 := 5e-12
echo := 0.0
ted := 24e-12
ln2x4 := 2.772588722239781
rho_e := 81e-8
C_vol := 2.6e6
d_stack := 16e-9
tauC := C_vol * d_stack / 170e6
Tcur := Troom
SetSolver(4)
FixDt = 5e-14
Jcur := 0.0
Ms_cur := Ms300
TableAddVar(Tcur, "T", "K")
TableAddVar(Jcur, "J", "A/m^2")
TableAddVar(Ms_cur, "Ms", "A/m")
TableAutoSave(0.5e-12)
OutputFormat = OVF2_BINARY
AutoSave(m, 25e-12)
m = Uniform(0, 0, 1)
relax()
saveas(m, "m_initial")
print("relaxed m = ", m.average())
//relaxed m =  [-0.1990163671875 1.294538378715515e-12 0.9799940625]
J = vector(0, 0, Isign*Jpk*(exp(-ln2x4*pow((t-t0)/tp, 2))+echo*exp(-ln2x4*pow((t-t0-ted)/tp, 2))))
tmax := 400e-12
stp := 0.25e-12
niter := 1600
tprev := 0.0
for i := 0; i < niter; i++ {
	tb := t
	Jcur = Isign * Jpk * (exp(-ln2x4*pow((tb-t0)/tp, 2)) + echo*exp(-ln2x4*pow((tb-t0-ted)/tp, 2)))
	dtr := tb - tprev
	heat := rho_e * Jcur * Jcur / C_vol
	Tcur = (Tcur + heat*dtr + Troom*dtr/tauC) / (1 + dtr/tauC)
	fr := 1.0 - pow(Tcur/Tc, 1.7)
	fr = 0.5 * (fr + abs(fr))
	Ms_cur = Ms0 * fr
	Msat = Ms_cur
	Ku1 = Kz0 * pow(fr, 3.0)
	tprev = tb
	run(stp)
}
J = vector(0, 0, 0)
Jcur = 0.0
tableSave()
saveas(m, "m_final")
mzavg := m.average().Z()
print("final mz = ", mzavg)
//final mz =  0.970219375
if mzavg < -0.5 {
	print("result: switched (+Mz -> -Mz)")
} else {
	print("result: NOT switched (raise Jpk, or check Hx and Isign)")
}
