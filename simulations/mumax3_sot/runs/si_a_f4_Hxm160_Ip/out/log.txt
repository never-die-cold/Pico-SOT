//output directory: E:\_SOT_MARM\simulations\mumax3_sot\runs\si_a_f4_Hxm160_Ip\out/
SetGridSize(64, 64, 1)
SetCellSize(16e-9, 16e-9, 1e-9)
// WARNING: cell size was set to a high aspect ratio.
//          Calculation of demag kernel may take longer than usual.
//          Note that this can exacerbate discretization anisotropy.
SetPBC(16, 16, 0)
//resizing...
Ms300 := 1.0e6
BK300 := 0.8
Kz300 := 0.5 * Ms300 * (BK300 + mu0*Ms300)
Troom := 300.0
Tc := 800.0
fr300 := 1.0 - pow(Troom/Tc, 1.7)
Ms0 := Ms300 / fr300
Kz0 := Kz300 / pow(fr300, 3.0)
Msat = Ms300
Aex = 3e-11
alpha = 0.23
Ku1 = Kz300
AnisU = vector(0, 0, 1)
DisableZhangLiTorque = true
Pol = 0.20
EpsilonPrime = 0.05
Lambda = 1
FreeLayerThickness = 1e-9
FixedLayer = vector(0, -1, 0)
J = vector(0, 0, 0)
Hx := -0.160
Isign := 1
B_ext = vector(Hx, 0, 0)
Jpk := 4.0e12
tp := 3.7e-12
t0 := 5e-12
echo := 0.3
ted := 24e-12
ln2x4 := 2.772588722239781
rho_e := 81e-8
C_vol := 2.6e6
d_stack := 16e-9
tauC := C_vol * d_stack / 170e6
SetSolver(4)
FixDt = 5e-14
Tcur := Troom
Jcur := 0.0
Ms_cur := Ms300
TableAddVar(Tcur, "T", "K")
TableAddVar(Jcur, "J", "A/m^2")
TableAddVar(Ms_cur, "Ms", "A/m")
TableAutoSave(0.25e-12)
OutputFormat = OVF2_BINARY
m = Uniform(0, 0, 1)
relax()
saveas(m, "m_initial")
print("relaxed m = ", m.average(), "  B_demag = ", B_demag.average())
//relaxed m =  [-0.19958430528640747 -1.2792505668321203e-15 0.979880690574646]   B_demag =  [1.2808487554138992e-05 -1.0254497438024596e-15 -1.229784607887268]
J = vector(0, 0, Isign*Jpk*(exp(-ln2x4*pow((t-t0)/tp, 2))+echo*exp(-ln2x4*pow((t-t0-ted)/tp, 2))))
tmax := 400e-12
stp := 0.25e-12
niter := 1600
tprev := 0.0
for i := 0; i < niter; i++ {
	tb := t
	Jcur = Isign * Jpk * (exp(-ln2x4*pow((tb-t0)/tp, 2)) + echo*exp(-ln2x4*pow((tb-t0-ted)/tp, 2)))
	dtr := tb - tprev
	heat := rho_e * Jcur * Jcur / C_vol
	Tcur = (Tcur + heat*dtr + Troom*dtr/tauC) / (1 + dtr/tauC)
	fr := 1.0 - pow(Tcur/Tc, 1.7)
	fr = 0.5 * (fr + abs(fr))
	Ms_cur = Ms0 * fr
	Msat = Ms_cur
	Ku1 = Kz0 * pow(fr, 3.0)
	tprev = tb
	run(stp)
}
J = vector(0, 0, 0)
Jcur = 0.0
tableSave()
saveas(m, "m_final")
print("final m = ", m.average())
//final m =  [-0.20224690437316895 -8.050898759393021e-05 0.9793344736099243]
